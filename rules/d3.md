# D3.js — Animation patterns

## When to use

Data visualization with motion — chart transitions, axis animations, data join enter/exit. D3's strength is binding data to DOM and transitioning between data states.

## Core pattern: data join with transition

```js
import * as d3 from "d3";

const svg = d3.select("svg");

function update(data) {
  const circles = svg.selectAll("circle").data(data, (d) => d.id);

  // EXIT: remove old elements
  circles.exit()
    .transition().duration(500)
    .attr("r", 0)
    .remove();

  // ENTER: create new elements
  circles.enter()
    .append("circle")
    .attr("cy", (d) => d.y)
    .attr("cx", (d) => d.x)
    .attr("r", 0)
    .transition().duration(500)
    .attr("r", (d) => d.r);

  // UPDATE: transition existing
  circles
    .transition().duration(500)
    .attr("cx", (d) => d.x)
    .attr("cy", (d) => d.y)
    .attr("r", (d) => d.r)
    .attr("fill", (d) => d.color);
}
```

## join() shorthand (D3 v5+)

```js
svg.selectAll("circle")
  .data(data)
  .join(
    enter => enter.append("circle").attr("r", 0)
      .call(enter => enter.transition().duration(500).attr("r", d => d.r)),
    update => update
      .call(update => update.transition().duration(500).attr("fill", d => d.color))
  );
```

## Line chart transition

```js
const line = d3.line().x(d => x(d.date)).y(d => y(d.value));

svg.append("path")
  .datum(data)
  .attr("d", line)
  .attr("fill", "none")
  .attr("stroke", "steelblue");

// Transition to new data
svg.selectAll("path")
  .datum(newData)
  .transition().duration(1000)
  .attr("d", line);
```

## Staggered bar chart

```js
svg.selectAll("rect")
  .data(data)
  .join("rect")
  .attr("x", d => x(d.name))
  .attr("width", x.bandwidth())
  .attr("y", height)
  .attr("height", 0)
  .transition()
  .duration(600)
  .delay((d, i) => i * 50)  // stagger
  .attr("y", d => y(d.value))
  .attr("height", d => height - y(d.value));
```

## Axis animation

```js
// Update scale, then transition axis
xScale.domain(newDomain);

svg.select(".x-axis")
  .transition().duration(500)
  .call(d3.axisBottom(xScale));
```

## Easing

```js
.transition()
.duration(800)
.ease(d3.easeCubicInOut)   // built-in
// .ease(d3.easeElasticOut.amplitude(1).period(0.3))
```

## React + D3 pattern

```tsx
import { useRef, useEffect } from "react";
import * as d3 from "d3";

function Chart({ data }) {
  const ref = useRef(null);

  useEffect(() => {
    const svg = d3.select(ref.current);
    // ... draw / update

    return () => {
      svg.selectAll("*").remove(); // cleanup
    };
  }, [data]);

  return <svg ref={ref} width={600} height={400} />;
}
```

## Performance tips

- Use Canvas renderer for 1000+ elements
- `.transition()` is per-element — 1000 circles = 1000 timers. Batch with `requestAnimationFrame` for large n
- Key function in `.data(data, key)` avoids unnecessary DOM manipulation
